\section{Kiến Trúc Server Và Game}

% ---------- Slide 1 ----------
\begin{frame}{Kiến Trúc TCP Server}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Cách server hoạt động}
        \begin{itemize}
          \item \texttt{TcpListener} nhận kết nối, \textbf{mỗi client một Task} (async/await).
          \item \texttt{ClientSession}: đọc gói theo \textbf{framing 4-byte}, gửi có khóa chống xen byte.
          \item \texttt{PacketRouter}: mỗi loại gói có một \textbf{handler} đăng ký riêng.
          \item Danh sách người online dùng \texttt{ConcurrentDictionary} (an toàn đa luồng).
        \end{itemize}
      \end{block}
      \footnotesize %%% ĐỨC KIỂM TRA: có muốn thêm ý nào về xử lý ngắt kết nối đột ngột không?

\begin{frame}{Quản Lý Phòng Và State Machine}
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \begin{block}{\texttt{RoomManager}}
        \begin{itemize}
          \item \texttt{ConcurrentDictionary} chứa các phòng.
          \item \textbf{Khóa riêng từng phòng} (\texttt{lock}) chống 2 người vào cùng lúc.
          \item Chủ phòng mới được \textbf{kick}.
          \item \textbf{Server là bên quyết định} (server-authoritative).
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.56\textwidth}
      % Sơ đồ trạng thái. Nét liền = ĐÃ code. Nét đứt = đã thiết kế, đang phát triển.
