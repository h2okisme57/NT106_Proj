\section{Giao diện Client & Tương tác CSDL}

% ── Frame 1: Đăng nhập ───────────────────────────────────────────
\begin{frame}{1. Giao Diện Đăng Nhập & Xác Thực}
\begin{columns}
\begin{column}{0.55\textwidth}
\begin{block}{Cơ chế xác thực}
\begin{itemize}
\item \textbf{Gửi Request:} Client đóng gói Username và Password thành \texttt{LoginRequestDto} gửi qua TCP.
\item \textbf{Truy vấn Database:} Server nhận gói tin, truy vấn CSDL để đối chiếu tài khoản.
\item \textbf{Cập nhật Profile:} Nhận \texttt{LoginResponse}, khởi tạo \texttt{PlayerProfile} trên máy khách.
\end{itemize}
\end{block}
\end{column}

\begin{frame}{2. Bảo Mật CSDL: Chống Spam Login}
\begin{columns}
\begin{column}{0.55\textwidth}
\begin{exampleblock}{Cơ chế Lockout (Khóa tạm thời)}
\begin{itemize}
\item \textbf{Giới hạn thử lại:} Nhập sai mật khẩu liên tiếp \textbf{5 lần}.
\item \textbf{Xử lý tại Client:} Vô hiệu hóa nút bấm và form nhập liệu (TextBox).
\item \textbf{Trải nghiệm UX:} Đếm ngược trực tiếp 60 giây trên giao diện cảnh báo lỗi.
\item \textbf{Ý nghĩa:} Giảm tải truy vấn rác xuống CSDL, chống Brute-force/Spam.
\end{itemize}
\end{exampleblock}
\end{column}

\begin{frame}{3. Sảnh Chờ (Lobby) & Ghép Trận}

\begin{column}{0.5\textwidth}
\begin{block}{Quản lý trạng thái người chơi}

\item \textbf{Danh sách Online:} Truy vấn và đồng bộ danh sách người chơi đang hoạt động từ Server.
\item \textbf{Thông tin User:} Hiển thị Tên, Cấp độ, Số trận thắng/thua được tải thẳng từ CSDL.
\item \textbf{Tương tác trực tiếp:} Gửi lời mời thách đấu (Match Request) dựa trên socket ID hiện tại.
\end{itemize}
\end{block}

% ── Frame 4: Bảng xếp hạng ───────────────────────────────────────────
\begin{frame}{4. Thống Kê & Bảng Xếp Hạng}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{itemize}
\item \textbf{Lưu trữ kết quả:} Sau mỗi trận đấu, Server tiến hành \texttt{UPDATE} bản ghi CSDL (cộng điểm, số trận thắng).
\item \textbf{Trích xuất top:} Dùng câu lệnh \texttt{ORDER BY Elo DESC} để lấy top những người chơi xuất sắc nhất.
\item \textbf{Minh bạch:} Dữ liệu cập nhật theo thời gian thực (Real-time).
\end{itemize}
\vspace{0.3cm}
\textit{* Lưu ý: Nếu dữ liệu trắng, Leaderboard sẽ trống. Các dữ liệu mẫu (Seeding) do Supabase đảm nhiệm.}

\section{Hạ tầng Đám mây & Web Service}

% ── Frame 5: Cấu trúc Database trên Cloud ────────────────────────────
\begin{frame}{5. Cơ Sở Dữ Liệu Đám Mây (Supabase)}
\begin{columns}
\begin{column}{0.55\textwidth}
\begin{block}{Chuyển đổi lên Cloud}
\begin{itemize}
\item \textbf{Triển khai:} Toàn bộ CSDL được host trên hạ tầng của Supabase (PostgreSQL) thay vì Localhost.
\item \textbf{Tổ chức bảng (Tables):} Cấu trúc chuẩn hóa với các bảng \texttt{users}, \texttt{match_history} để quản lý hồ sơ và lịch sử đấu.
\item \textbf{Làm việc nhóm:} Dễ dàng cấp quyền (Invite) cho các thành viên trong nhóm cùng quản lý dữ liệu.
\end{itemize}
\end{block}
\end{column}

% ── Frame 6: Bảo mật REST API (Chặn truy cập trái phép) ────────────────
\begin{frame}{6. Bảo Mật REST API (Row Level Security)}
\begin{columns}
\begin{column}{0.55\textwidth}
\begin{alertblock}{Ngăn chặn truy cập trái phép}
\begin{itemize}
\item \textbf{Nguy cơ:} Nếu API Public, bất kỳ ai cũng có thể gọi HTTP GET/POST để xem hoặc sửa điểm số, mật khẩu.
\item \textbf{Bảo vệ:} Kích hoạt RLS (Row Level Security) trên Supabase, chặn mọi truy cập nặc danh (Anonymous).
\item \textbf{Kiểm thử:} Chạy script \texttt{demo_rest.ps1} không đính kèm Service Key, máy chủ lập tức từ chối (HTTP 401 Unauthorized).
\end{itemize}
\end{alertblock}
\end{column}
\begin{column}{0.45\textwidth}
\centering
\slideimg[width=\linewidth,height=0.65\textheight,keepaspectratio]{02-rest-blocked.png}{Lỗi truy cập REST API khi thiếu Key}
\end{column}
\end{columns}
\end{frame}

% ── Frame 7: REST API (Truy xuất thành công) ───────────────────────
\begin{frame}{7. Truy Xuất Dữ Liệu Qua REST API}
\begin{columns}
\begin{column}{0.55\textwidth}
\begin{exampleblock}{Giao tiếp HTTP & JSON}
\begin{itemize}
\item \textbf{Xác thực Header:} Đính kèm \texttt{apikey} và \texttt{Authorization: Bearer} vào HTTP Header để định danh.
\item \textbf{Định dạng trả về:} Dữ liệu từ PostgreSQL được trả về dưới dạng \textbf{JSON} gọn nhẹ.
\item \textbf{Tích hợp C#:} Server dùng \texttt{HttpClient} để parse chuỗi JSON này thành các object (như \texttt{UserDbModel}), từ đó đẩy Bảng xếp hạng (Leaderboard) về Client.
\end{itemize}
\end{exampleblock}
\end{column}
